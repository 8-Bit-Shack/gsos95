# Baut START.GS.OS mit GS/OS95-Bootbild aus der Original-Datei
import sys, struct
BASE = 0x6800
def off(a): return a - BASE
def patch(orig):
    d = bytearray(orig)
    def chk(a, b):
        assert d[off(a):off(a)+len(b)] == bytes(b), 'unerwartet bei %X: %s' % (a, d[off(a):off(a)+len(b)].hex())
    def put(a, b): d[off(a):off(a)+len(b)] = bytes(b)
    chk(0x7393, [0x20, 0xF4, 0x75]); chk(0x7396, [0x20, 0x0E, 0x76]); chk(0x7399, [0x20, 0x2D, 0x76])
    chk(0x76BB, [0xA9, 0x49, 0x57]); chk(0x76CC, [0xA0, 0x23, 0x00]); chk(0x76E4, [0xA0, 0x23, 0x00])
    chk(0x7505, [0xA9, 0xE9, 0x57]); chk(0x7515, [0xC0, 0x24, 0x00]); chk(0x75F4, [0xA9, 0xC5, 0x43])
    chk(0x7B71, [0x7F, 0xFF, 0x00] if False else d[off(0x7B71):off(0x7B71)+3])
    code = open(__import__('os').path.join(__import__('os').path.dirname(__file__), 'boot95.bin'), 'rb').read()
    assert 0x7778 + len(code) <= 0x7B71, len(code)
    put(0x7778, code)
    put(0x7638, code2())
    put(0x75F4, [0x4C, 0x78, 0x77])                 # JMP $7778 (neues Bootbild)
    chk(0x7485, [0x20, 0xF3, 0x74]); put(0x7485, [0x20, 0x7B, 0x77])   # zwei Schritte je Datei
    assert code[3:6] == bytes([0x20, 0xF3, 0x74])
    # Balkenmuster: Bloecke aus 6 weissen Punkten + 2 Punkten Luecke (statt Streifen)
    chk(0x756E, bytes.fromhex('7fff4ffff7fff4ffff7fff4ffff7fff4'))
    put(0x756E, bytes.fromhex('7fff5ffff7fff5ffff7fff5fffffffff'))
    put(0x7396, [0xEA] * 3); put(0x7399, [0xEA] * 3) # Symbol/Text entfallen
    # Fortschrittsbalken: Zeile 166, ab Byte 30, 101 Bytes breit
    fb = 0x2000 + 166 * 160 + 30
    put(0x76BC, struct.pack('<H', fb))
    put(0x7506, struct.pack('<H', fb + 0xA0))
    put(0x76CD, [0x63]); put(0x76E5, [0x63]); put(0x7516, [0x64])
    # Kontrollwort "Bootbild noch sichtbar": unterer Balkenrahmen (schwarz = $0000)
    chk(0x74E0, [0xAF, 0x65, 0x44, 0xE1, 0xC9, 0xF0, 0x00])
    sent = 0x2000 + 171 * 160 + 40
    put(0x74E1, struct.pack('<H', sent)); put(0x74E5, [0x00, 0x00])
    return bytes(d)
def code2():
    import os
    c = open(os.path.join(os.path.dirname(__file__), 'boot95b.bin'), 'rb').read()
    assert 0x7638 + len(c) <= 0x76BB, len(c)   # alter, nicht mehr benutzter Code
    return c
def repatch(done):
    """Neues Bootbild in eine bereits gepatchte START.GS.OS einsetzen."""
    d = bytearray(done)
    assert d[0x75F4 - BASE:0x75F7 - BASE] == bytes([0x4C, 0x78, 0x77])
    code = open(__import__('os').path.join(__import__('os').path.dirname(__file__), 'boot95.bin'), 'rb').read()
    assert 0x7778 + len(code) <= 0x7B71, len(code)
    d[0x7778 - BASE:0x7778 - BASE + len(code)] = code
    c2 = code2()
    d[0x7638 - BASE:0x7638 - BASE + len(c2)] = c2
    return bytes(d)
if __name__ == '__main__' and len(sys.argv) > 1 and sys.argv[1] == '-r':
    open(sys.argv[3], 'wb').write(repatch(open(sys.argv[2], 'rb').read())); print('ok'); sys.exit()
if __name__ == '__main__':
    o = open(sys.argv[1], 'rb').read()
    open(sys.argv[2], 'wb').write(patch(o))
    print('ok')
