; GS/OS95-Bootbild: ersetzt in START.GS.OS das Zeichnen von Kasten, Symbol
; und Text (Aufruf ueber JMP bei $75F4). Laeuft in Bank 0 ab $7778.
.setcpu "65816"
.smart off
.a16
.i16
; Variablen im nicht mehr benutzten Code des Originals ($760E-$764A)
CNT = $760E
POFF = $7610
ROW = $7612
COLV = $7614
SPP = $7616
SXV = $7618
SYV = $761A
CLO = $761C
CHI = $761E
W = $7620
H = $7622
BPR = $7624
DPT = $7626
RR = $7628
T = $762A
II = $762C
BB = $762E
SCL = $7630
YY = $7632
SY2 = $7634
XX = $7636
        .segment "CODE"
start:  jmp main
; Ladebalken: zwei Schritte pro geladener Datei (Balken ist breiter als im Original)
step2:  jsr $74F3
        jmp $74F3
main:   php
        phb
        phk
        plb
        rep #$30
        ; ---- Paletten: 0 = Balken (640er-Modus), 1..14 = Flaeche + Schrift,
        ; 15 = nur Flaeche (unten: Symbole der Systemerweiterungen unsichtbar).
        ; Das Bild selbst (Paletten 1..13 und Zeilen 0..165) setzt danach
        ; SPLASH95 aus *:System:GSOS95.SPL ein ----
        ldx #30
@p0:    lda d_pal0,x
        sta f:$E19E00,x
        lda d_p14,x
        sta f:$E19E00+14*32,x
        lda #BANDC
        sta f:$E19E00+15*32,x
        dex
        dex
        bpl @p0
        ldx #13*32-2
@pn:    txa                     ; Palette 14 auch nach 1..13

        and #$001F
        tay
        lda d_p14,y
        sta f:$E19E00+32,x
        dex
        dex
        bpl @pn
        ; ---- Zeilen: SCB und Fuellung ----
        ldx #0
        stz ROW
@fr:    lda ROW
        cmp #BAR0
        bcc @g
        cmp #BAR1+1
        bcs @u
        lda #$FFFF              ; Balkenzeilen: 640er-Modus, Wert 3 = Flaeche
        sta COLV
        lda #$0080
        bra @s
@u:     stz COLV                ; unten: Palette 15
        lda #15
        bra @s
@g:     stz COLV                ; Flaeche: Palette 14, Farbe 0
        lda #14
@s:     phx
        ldx ROW
        sep #$20
        .a8
        sta f:$E19D00,x
        rep #$20
        .a16
        plx
        lda COLV
        ldy #80
@fw:    sta f:$E12000,x
        inx
        inx
        dey
        bne @fw
        inc ROW
        lda ROW
        cmp #200
        bcc @fr
        ; ---- Pixel-Schriftzug GS/OS 95 (weisser Schatten, dunkelblau) ----
        lda #4
        sta SCL
        lda #2
        sta COLV
        lda #d_logo
        ldx #LOGO_X+3
        ldy #LOGO_Y+3
        jsr blit
        lda #1
        sta COLV
        lda #d_logo
        ldx #LOGO_X
        ldy #LOGO_Y
        jsr blit
        ; ---- "Created by ..." und Build wie im Startbild ----
        lda #1
        sta SCL
        sta COLV
        lda #d_by
        ldx #BY_X
        ldy #BY_Y
        jsr blit
        lda #d_build
        ldx #BUILD_X
        ldy #BUILD_Y
        jsr blit
fin:    lda #$00E1
        sta $02
        plb
        plp
        rts

; 1-Bit-Sprite A an X,Y in Farbe COLV
blit:   sta SPP
        stx SXV
        sty SYV
        lda COLV
        sta CLO
        asl
        asl
        asl
        asl
        sta CHI
        ldx SPP
        lda a:0,x
        and #$00FF
        sta W
        lda a:1,x
        and #$00FF
        sta H
        lda W
        clc
        adc #7
        lsr
        lsr
        lsr
        sta BPR
        inx
        inx
        stx DPT
        stz RR
@row:   lda RR
        ldx SCL
        jsr mul
        clc
        adc SYV
        sta YY
        stz SY2
@sub:   lda YY
        clc
        adc SY2
        asl
        asl
        asl
        asl
        asl
        sta T
        asl
        asl
        clc
        adc T
        clc
        adc #$2000
        sta $00
        lda #$00E1
        sta $02
        stz II
@col:   lda II
        lsr
        lsr
        lsr
        clc
        adc DPT
        tax
        lda a:0,x
        and #$00FF
        sta BB
        lda II
        and #7
        tax
        lda BB
@sh:    cpx #0
        beq @g
        asl
        dex
        bra @sh
@g:     and #$0080
        beq @nx
        lda II
        ldx SCL
        jsr mul
        clc
        adc SXV
        sta XX
        ldx SCL
@px:    phx
        lda XX
        lsr
        tay
        sep #$20
        .a8
        bcs @odd
        lda [$00],y
        and #$0F
        ora CHI
        sta [$00],y
        bra @d
@odd:   lda [$00],y
        and #$F0
        ora CLO
        sta [$00],y
@d:     rep #$20
        .a16
        inc XX
        plx
        dex
        bne @px
@nx:    inc II
        lda II
        cmp W
        bcs *+5
        jmp @col
        inc SY2
        lda SY2
        cmp SCL
        bcs *+5
        jmp @sub
        lda DPT
        clc
        adc BPR
        sta DPT
        inc RR
        lda RR
        cmp H
        bcs *+5
        jmp @row
        rts

; A * X -> A
mul:    sta T
        lda #0
@m:     cpx #0
        beq @r
        clc
        adc T
        dex
        bra @m
@r:     rts

        .include "data.inc"
endmark:
; Build-Zeile im nicht mehr benutzten Code des Originals ($7638-$76BA)
        .segment "DATA2"
        .include "data2.inc"
